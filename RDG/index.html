UL Render Dependency Graphs

January 23, 2026

This describes learning RDG by converting this FFT Base Ocean Simulation which is implemented in Niagara to RDG.

This is for the purpose of learning RDG, not because I have any opinion about where RGD is faster/slower than Niagara, or more suited to different kinds of projects or usages.

Fundamental Approach​

In the FFT Waves Example we have:

  • a plane mesh object (1,025 x 1,025 vertices = 1,050,625 vertices)
  • a Niagara simulation
  • a set of render targets which holds vertex and pixel data written to by the Niagara simulation. These are much lower resolution that the plane mesh (say 256 x 256)
  • a material used on the plane mesh which accesses the render targets and calculates vertical and horizontal displacement at each of the mesh vertices

The Niagara System​

Broadly speaking the Niagara system has one emitter, which holds stages which configure the emitter and sometimes execute shaders on the GPU. Each of these stages is described below:

#if GPU_SIMULATION

// Retrive WaveVector from thread index. 
int2 Position = int2(GDispatchThreadId.x,GDispatchThreadId.y);
float2 WaveVector = float2(Position.x-HalfGridSize, Position.y-HalfGridSize);
WaveVector *= 2.0f * PI;

//Our parameters for each of cascade are stored as components of a 4D vector.
//We access them, treating 4D vector as array of 4 scalars,
//And our Z component of thread index is the index of cascade, current compute thread belongs to.
WaveVector /= PatchLength[GDispatchThreadId.z];

// Calculate magnitude of WaveVector
float k = length(WaveVector);

//Result is magnitude of positive and negative spectrums
float2 Result =float2(0,0);

// Handle zero length case
if (k>0.0001f)
{
    // Calculate direction of wave vector
    float2 k_norm = normalize(WaveVector);

    // Calculate dampening factor for wave directions not aligning with the wind
    // Positive spectrum is in R component, negative spectrum is in G component.
    float2 WindFactor; 
    WindFactor.x = dot(k_norm, WindDirection);
    WindFactor.y = dot(-k_norm, WindDirection);
    float2 WindFactorAbs = abs( WindFactor);
    WindFactorAbs = pow( WindFactorAbs, WindTighten[GDispatchThreadId.z] );

    // Reduce magnitude of the waves, travelling in negative direction
    WindFactorAbs.x *= WindFactor.x >0 ? 1: WindDirectionality[GDispatchThreadId.z];
    WindFactorAbs.y *= WindFactor.y >0 ? 1: WindDirectionality[GDispatchThreadId.z];
    WindFactor = WindFactorAbs;

    // Phillips Ocean spectrum calculation
    float L = WindSpeed * WindSpeed / g;
    float UpperPart = exp(-1.0f / ((k*L) * (k*L)));
    float Spectrum = A[GDispatchThreadId.z] * UpperPart / (k * k * k * k);

    // Dampen waves, shorter than user controlled threshold
    Spectrum*=exp(-(k * k) * ShortWaveCutoff[GDispatchThreadId.z]);
    Spectrum *= k < LongWaveCutoff[GDispatchThreadId.z] ? 0 : 1; 

    // Only wind factor is different between positive and negative spectrums
    Result=sqrt(Spectrum*WindFactor);
    Result*=1.0f/sqrt(2.0f);
}

// Lastly, obtain complex amplitude/phases.
float4 PositiveAndNegativeSpectrum=float4(Result.rr,Result.gg);
PositiveAndNegativeSpectrum *= ComplexAmplitudes;

// And obtain conjugate of negative spectrum by flipping its imaginary part. 
PositiveAndNegativeSpectrum.w *= -1;

// Prepare index of the grid, at which result will be stored and write the result.
int2 IndexXY=int2(Position.x,Position.y+GDispatchThreadId.z*GridSize);
SpectrumGrid.SetVector4Value<Attribute="Spectrum">(IndexXY.x, IndexXY.y, PositiveAndNegativeSpectrum);

#endif //gpu simulation

#if GPU_SIMULATION

// Retrieve WaveVector from thread index
float2 WaveVector = float2(int(GDispatchThreadId.x)-HalfGridSize, int(GDispatchThreadId.y)-HalfGridSize);
bool bSkip = ((GDispatchThreadId.x == 256) && (GDispatchThreadId.y == 256));
WaveVector *= 2.0f * PI;
WaveVector /= PatchLength[GDispatchThreadId.z];

// Calculate magnitude of WaveVector
float k = length(WaveVector);

// Dispersion. Angular frequency from wave vector
float Freq = sqrt(g * k);

// Quantize frequency to multiple of base frequency
Freq = floor(Freq / BaseFreq) * BaseFreq;

float Phase = Freq * AnimationTime;

// Load initial spectrum
float4 h0;
int IndexY = GDispatchThreadId.y + GDispatchThreadId.z * GridSize;
SpectrumGrid.GetPreviousVector4Value<Attribute="Spectrum">(GDispatchThreadId.x, IndexY, h0);

// Calculate exponents for positive and negative spectrums
float2 fourier_amp = h0.rg;
float2 fourier_amp_conj=h0.ba;
float2 SineCosine;
sincos(Phase, SineCosine.x,SineCosine.y);

float2 exponent = float2(SineCosine.y, SineCosine.x);
float2 exponent_inv = float2(SineCosine.y, -SineCosine.x);

// Resulting displacements
float2 DispZ=float2(0,0);
float2 DispX=float2(0,0);
float2 DispY=float2(0,0);

//Handle zero length case.
if (k>0.000001f)
{
    //Complex multiplication of positive spectrum by exponent
    float2 c0 = jMul(fourier_amp, exponent);

    //Complex multiplication of negative spectrum by inverse exponent
    float2 c1 = jMul(fourier_amp_conj, exponent_inv);

    //Complex addition of positive and negative parts
    DispZ = jAdd(c0, c1);

    //Calculate horizontal displacements by projecting vertical displacement on components of wave direction
    float2 dx = float2(0.0f, WaveVector.x/k)*Chop[GDispatchThreadId.z];
    DispX = jMul(DispZ,dx);

    float2 dy = float2(0.0f,  WaveVector.y/k)*Chop[GDispatchThreadId.z];
    DispY = jMul(DispZ,dy);
}

//Store results into the grid
FFTGrid.SetVector4Value<Attribute="DX_DY">(GDispatchThreadId.x, IndexY, float4(DispX, DispY) );
FFTGrid.SetVector2DValue<Attribute="DZ">(GDispatchThreadId.x, IndexY, DispZ );
#endif// gpu simulation

The custom HLSL node includes file file OceanComplexMath.ush which has these functions:

float2 jMul(float2 c0,float2 c1)
{
	float2 c;
	c.x=c0.x*c1.x-c0.y*c1.y;
	c.y=c0.x*c1.y+c0.y*c1.x;
	return c;
}

float2 jAdd(float2 c0,float2 c1)
{
	float2 c;
	c.x=c0.x+c1.x;
	c.y=c0.y+c1.y;
	return c;
}

float jMod(float x,float y)
{
	return x-y*floor(x/y);
}

float2 jConj(float2 c)
{
	return float2(c.x,-c.y);
}

#define BUTTERFLY_COUNT 8
#define PI 3.1415926
//#define THREADGROUP_SIZE 256

void GetButterflyValues(uint passIndex,uint x, out uint2 indices, out float2 weights)
{
	int sectionWidth=2U<<passIndex;
	int halfSectionWidth=sectionWidth/2;
	
	int sectionStartOffset=x&~(sectionWidth-1);
	int halfSectionOffset=x&(halfSectionWidth-1);
	int sectionOffset=x&(sectionWidth-1);
	
	sincos(2.0*PI*sectionOffset/(float)sectionWidth,weights.y,weights.x);
	weights.y=-weights.y;
	
	indices.x=sectionStartOffset+halfSectionOffset;
	indices.y=sectionStartOffset+halfSectionOffset+halfSectionWidth;
	
	if (passIndex==0)
	{
		indices=reversebits(indices)>>(32-BUTTERFLY_COUNT)&(LENGTH-1);
	}
}
/////////////////
//groupshared float3 pingPongArray[4][THREADGROUP_SIZE];
groupshared float3 pingPongArray[4][LENGTH];

void ButterflyPass(int passIndex, uint x,uint t0, uint t1, out float3 resultR, out float3 resultI)
{
	uint2 Indices;
	float2 Weights;
	
	GetButterflyValues(passIndex,x,Indices,Weights);
	
	float3 inputR1=pingPongArray[t0][Indices.x];
	float3 inputI1=pingPongArray[t1][Indices.x];
	
	float3 inputR2=pingPongArray[t0][Indices.y];
	float3 inputI2=pingPongArray[t1][Indices.y];
	
	resultR=(inputR1+Weights.x*inputR2+Weights.y*inputI2)*0.5;
	resultI=(inputI1-Weights.y*inputR2+Weights.x*inputI2)*0.5;
}

void ButterflyPassFinalNoI(int passIndex,int x,int t0,int t1, out float3 resultR)
{
	uint2 Indices;
	float2 Weights;
	GetButterflyValues(passIndex,x,Indices,Weights);
	
	float3 inputR1=pingPongArray[t0][Indices.x];
	
	float3 inputR2=pingPongArray[t0][Indices.y];
	float3 inputI2=pingPongArray[t1][Indices.y];
	
	resultR=(inputR1+Weights.x*inputR2+Weights.y*inputI2)*0.5;
}

// Calculate size of 1 texture texel in meters
float WorldTexelSize =  (PatchLength[GDispatchThreadId.z]) /256.0f;

// Elements of Jacobian matrix
float Jxx=1+(R.x-C.x) /WorldTexelSize  ;
float Jyy=1+(B.y-C.y) /WorldTexelSize ;

float Jyx=(R.y-C.y) /WorldTexelSize  ;
float Jxy=(B.x-C.x) /WorldTexelSize  ;
// Determinant of Jacobian matrix
float JacobianDeterminant=Jxx*Jyy-Jxy*Jxy;
// Bias Jacobian determinant by user parameter
JacobianDeterminant =  - JacobianDeterminant + FoamThreshold[GDispatchThreadId.z]+1;

// Calculate J-
float EigenValue_ = (Jxx + Jyy) * 0.5f - pow( (Jxx-Jyy)*(Jxx-Jyy) + 4.0f * Jxy * Jxy , 0.5f ) * 0.5f;
	
// Calculate foam injection from eigenvalue and user parameter;
float Foam=FoamMultiplier[GDispatchThreadId.z]*saturate(+1-EigenValue_+ FoamThreshold[GDispatchThreadId.z] );

// Load previous frame foam at current position
float PreviousFoamCenter=0;
SamplePos=GetWrappedPosition(position.xyz,int2(0,0));
FoamGrid.GetPreviousFloatValue<Attribute="Foam">(SamplePos.x, SamplePos.y, PreviousFoamCenter);
	
// Load previous frame foam at top, bottom, right and left texels and add result to accumulated foam.
float AccumulatedFoam=0;
for(int q=1; q<5; q++)
{
    SamplePos=GetWrappedPosition(position.xyz,SampleOffsets[q]);
    float FoamSample;
    FoamGrid.GetPreviousFloatValue<Attribute="Foam">(SamplePos.x, SamplePos.y, FoamSample);
    AccumulatedFoam += FoamSample;
}

// interpolate from unblurrred previous foam to fully blurred previous foam using delta time and user parameter
float PreviousFoam=lerp(PreviousFoamCenter,AccumulatedFoam*0.25f,saturate(DeltaTime * FoamBlur[GDispatchThreadId.z]));
	
// Decrease previous foam by user parameter and delta time
float FadedPreviousFoam=saturate(PreviousFoam-FoamFade[GDispatchThreadId.z]*DeltaTime);
	
// Add this frame's foam to previous foam
Foam = max(Foam, FadedPreviousFoam);

// Calculate derivative of X displacement along X axis
float DXx=R.x-C.x;
	
// Calculate derivative of Z displacement along X axis
float DZx=(R.z-C.z);
	
// Calculate derivative of Y displacement along Y axis
float DYy=B.y-C.y;
	
// Calculate derivative of Z displacement along Y axis
float DZy=(B.z-C.z);
	
// Calculate derivative of X displacement along Y axis
float DXy=(B.x-C.x);
	
// Calculate derivative of Y displacement along X axis
float DYx=(R.y-C.y);
	
// Update previous foam and previous displacements in foam grid to new values
FoamGrid.SetFloatValue<Attribute="Foam">(position.x, position.y+GDispatchThreadId.z*256, Foam);
FoamGrid.SetVectorValue<Attribute="PreviousDisp">(position.x, position.y+GDispatchThreadId.z*256, C);

float4 AttribsA = float4(DXx,DZx,DYy,DZy)/WorldTexelSize;
float4 AttribsB = float4(DXy/WorldTexelSize,DYx/WorldTexelSize,Foam,JacobianDeterminant);

// Write out pixel attributes to Foam Grid
FoamGrid.SetVector4Value<Attribute="PixelAttribA">(position.x, position.y+GDispatchThreadId.z*256, AttribsA);
FoamGrid.SetVector4Value<Attribute="PixelAttribB">(position.x, position.y+GDispatchThreadId.z*256, AttribsB);

//Write out Vertex attributes render target arrays, converting from meters to centimeters. 
RT_VertexAttributes.SetRenderTargetValue(true, position.x, GDispatchThreadId.y, GDispatchThreadId.z, float4(C,Cprev.x)*100);
RT_VertexAttributesB.SetRenderTargetValue(true, position.x, GDispatchThreadId.y, GDispatchThreadId.z, float4(Cprev.yz*100,0,0));

#endif // gpu simulation
IGNORE=0;

#define LENGTH 256

int2 GetWrappedPosition(int3 pos,int2 offset)
{
	int2 posi=pos.xy+offset;
	FLATTEN
	if (posi.x>=LENGTH) posi.x-=LENGTH;
	FLATTEN
	if (posi.x<0) posi.x+=LENGTH;
	FLATTEN
	if (posi.y>=LENGTH) posi.y-=LENGTH;
	FLATTEN
	if (posi.y<0) posi.y+=LENGTH;

}

#if GPU_SIMULATION
#define SAMPLES_PER_FRAME 16
// Run a loop for number of integration samples per frame
float RoughnessSum = 0;
for(int FrameSamples=0; FrameSamples < SAMPLES_PER_FRAME; FrameSamples++)
{
    // Generate random position in the world. Assuming 0->999999999 range reasonably represents world
    float2 PositionA = float2(rand_float(999999999.0f),rand_float(999999999.0f));

    // LUT is going to cover range of 0 to this value, in world units. Hardcoded to about 20 meters
    float MaxRange = float(GDispatchThreadId.x) * 0.5f;

    //Generate second random position, uniformly distributed on a circle, within 4096 units from first position
    float sin_theta;
    float cos_theta;
    sincos(rand_float(PI*2.0f),sin_theta,cos_theta);
    float2 PositionB = PositionA + float2(sin_theta,cos_theta) * sqrt(rand_float(1.0f)) * MaxRange;

    // Convert PositionA from world space to texture space of each cascade
    float4 CascPosX = PositionA.xxxx / PatchLength / 100.0f;
    float4 CascPosY = PositionA.yyyy / PatchLength / 100.0f;

    //Sample each cascade derivatives and store values
    float4 SampleValues[4];
    RT_PixelAttribsA_Casc0.SampleRenderTargetValue(float2(CascPosX[0],CascPosY[0]), 0.0f, SampleValues[0]);
    RT_PixelAttribsA_Casc1.SampleRenderTargetValue(float2(CascPosX[1],CascPosY[1]), 0.0f, SampleValues[1]);
    RT_PixelAttribsA_Casc2.SampleRenderTargetValue(float2(CascPosX[2],CascPosY[2]), 0.0f, SampleValues[2]);
    RT_PixelAttribsA_Casc3.SampleRenderTargetValue(float2(CascPosX[3],CascPosY[3]), 0.0f, SampleValues[3]);

    //Gather derivatives from 4 cascades and weight them by cascade size;
    float3 DerivXSum = float3(1,0,0);
    float3 DerivYSum = float3(0,1,0);
    float4 TexelSizes = 4.0f * 256.0f / PatchLength;
    for(int i=0; i<4; i++)
    {
        DerivXSum += float3(SampleValues[i].r, 0, SampleValues[i].g);
        DerivYSum += float3(0, SampleValues[i].b, SampleValues[i].a);
    }

    //Obtain normal vector at first position from derivatives;
    float3 NormalA = normalize(cross(DerivXSum,DerivYSum));

    // Convert PositionB from world space to texture space of each cascade
    CascPosX = PositionB.xxxx / PatchLength / 100.0f;
    CascPosY = PositionB.yyyy / PatchLength / 100.0f;

    //Sample each cascade derivatives and store values
    RT_PixelAttribsA_Casc0.SampleRenderTargetValue(float2(CascPosX[0],CascPosY[0]), 0.0f, SampleValues[0]);
    RT_PixelAttribsA_Casc1.SampleRenderTargetValue(float2(CascPosX[1],CascPosY[1]), 0.0f, SampleValues[1]);
    RT_PixelAttribsA_Casc2.SampleRenderTargetValue(float2(CascPosX[2],CascPosY[2]), 0.0f, SampleValues[2]);
    RT_PixelAttribsA_Casc3.SampleRenderTargetValue(float2(CascPosX[3],CascPosY[3]), 0.0f, SampleValues[3]);

    //Gather derivatives from 4 cascades ;
    DerivXSum = float3(1,0,0);
    DerivYSum = float3(0,1,0);
    for(int q=0; q<4; q++)
    {
         DerivXSum += float3( SampleValues[q].r, 0, SampleValues[q].g);
         DerivYSum += float3(0,  SampleValues[q].b, SampleValues[q].a);
    }

    //Obtain normal vector at second position from derivatives;
    float3 NormalB = normalize(cross(DerivXSum,DerivYSum));

    // Express roughness as angle between pair of normals. Aligned->smooth ,Opposite->rough
    float Roughness = dot(NormalA,NormalB);
    Roughness*=0.5;
    Roughness+=0.5;

    RoughnessSum += Roughness;
} // end for number of samples per frame loop

RoughnessSum /= float(SAMPLES_PER_FRAME);

// Get previous frame roughness from grid
float PrevRoughness;
RoughnessGrid.GetPreviousFloatValue<Attribute="Roughness">(GDispatchThreadId.x, 0, PrevRoughness);

// Prepare weights for new sample and history
float SampleW = 1.0f / float(IntegrationSamples);
float PrevSampleW = 1.0f - SampleW;

// Invert roughness
float NewRoughness = 1.0f - saturate(RoughnessSum);

// Adjust roughness by user-controlled param
NewRoughness = pow(NewRoughness,RoughnessPower);

// Mix new roughness and history.
NewRoughness = PrevRoughness * PrevSampleW + NewRoughness * SampleW;

// Store roughness to grid and LUT render target;
RoughnessGrid.SetFloatValue<Attribute="Roughness">(GDispatchThreadId.x, 0, NewRoughness);
RT_Roughness.SetRenderTargetValue(true, GDispatchThreadId.x, 0, float4(NewRoughness, 0, 0, 0) );

#undef SAMPLES_PER_FRAME
#endif	 // gpu simulation
IGNORE=0;
